cmake_minimum_required(VERSION 3.22)
project(Epinizer VERSION 3.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(APPLE)
  set(CMAKE_OSX_DEPLOYMENT_TARGET "11.0" CACHE STRING "Minimum macOS")
  set(EPINIZER_FORMAT AU)
elseif(WIN32)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  set(EPINIZER_FORMAT VST3)
else()
  message(FATAL_ERROR "Epinizer ships AU on macOS and VST3 on Windows only")
endif()
include(CTest)
add_executable(epinizer_dsp_tests tests/DspTests.cpp)
target_include_directories(epinizer_dsp_tests PRIVATE Source)
add_test(NAME DSP COMMAND epinizer_dsp_tests)
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/JUCE/CMakeLists.txt")
  include(FetchContent)
  FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 29396c22c93392d6738e021b83196283d6e4d850 GIT_SHALLOW FALSE)
  FetchContent_MakeAvailable(JUCE)
else()
  add_subdirectory(third_party/JUCE)
endif()
if(CMAKE_CROSSCOMPILING)
  set(EPINIZER_AUTO_MANIFEST FALSE)
else()
  set(EPINIZER_AUTO_MANIFEST TRUE)
endif()
juce_add_plugin(Epinizer
  COMPANY_NAME "MyDarknessDays" BUNDLE_ID "audio.epinizer.effect"
  PLUGIN_MANUFACTURER_CODE Epna PLUGIN_CODE Epnz
  FORMATS ${EPINIZER_FORMAT} PRODUCT_NAME "Epinizer"
  IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE
  IS_MIDI_EFFECT FALSE COPY_PLUGIN_AFTER_BUILD FALSE
  VST3_CATEGORIES Fx Reverb Modulation Distortion
  VST3_AUTO_MANIFEST ${EPINIZER_AUTO_MANIFEST}
  AU_SANDBOX_SAFE TRUE
  AU_MAIN_TYPE kAudioUnitType_Effect)
target_sources(Epinizer PRIVATE Source/PluginProcessor.cpp Source/PluginEditor.cpp)
juce_add_binary_data(EpinizerAssets SOURCES Assets/castle-entrance.png)
target_compile_definitions(Epinizer PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(Epinizer PRIVATE EpinizerAssets juce::juce_audio_utils PUBLIC juce::juce_recommended_config_flags juce::juce_recommended_warning_flags)
juce_add_console_app(epinizer_plugin_tests PRODUCT_NAME "Epinizer integration tests")
# Shared plugin code exports JUCE_STANDALONE_APPLICATION=0; avoid a conflicting console definition.
get_target_property(EPINIZER_TEST_DEFS epinizer_plugin_tests COMPILE_DEFINITIONS)
list(REMOVE_ITEM EPINIZER_TEST_DEFS "JUCE_STANDALONE_APPLICATION=1")
set_target_properties(epinizer_plugin_tests PROPERTIES COMPILE_DEFINITIONS "${EPINIZER_TEST_DEFS}")
target_sources(epinizer_plugin_tests PRIVATE tests/PluginTests.cpp)
target_include_directories(epinizer_plugin_tests PRIVATE Source)
target_link_libraries(epinizer_plugin_tests PRIVATE Epinizer juce::juce_audio_utils)
add_test(NAME PluginState COMMAND epinizer_plugin_tests)
set_tests_properties(DSP PluginState PROPERTIES TIMEOUT 60)
add_executable(epinizer_render_demo tools/RenderDemo.cpp)
target_include_directories(epinizer_render_demo PRIVATE Source)
if(WIN32)
  if(CMAKE_CROSSCOMPILING)
    # Pinned JUCE helper: generate metadata with Wine after cross compilation.
    _juce_add_vst3_manifest_helper_target(Epinizer)
  endif()
  juce_add_console_app(epinizer_vst3_host_tests PRODUCT_NAME "Epinizer VST3 host tests")
  target_sources(epinizer_vst3_host_tests PRIVATE tests/Vst3HostTests.cpp)
  target_compile_definitions(epinizer_vst3_host_tests PRIVATE JUCE_PLUGINHOST_VST3=1 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
  target_link_libraries(epinizer_vst3_host_tests PRIVATE juce::juce_audio_utils)
  add_test(NAME VST3Host COMMAND epinizer_vst3_host_tests "$<TARGET_FILE_DIR:Epinizer_VST3>/../..")
  set_tests_properties(VST3Host PROPERTIES TIMEOUT 60)
endif()
add_executable(epinizer_character_tests tests/CharacterTests.cpp)
target_include_directories(epinizer_character_tests PRIVATE Source)
add_test(NAME Characters COMMAND epinizer_character_tests)
set_tests_properties(Characters PROPERTIES TIMEOUT 60)
add_executable(epinizer_realtime_tests tests/RealtimeTests.cpp)
target_include_directories(epinizer_realtime_tests PRIVATE Source)
add_test(NAME Realtime COMMAND epinizer_realtime_tests)
set_tests_properties(Realtime PROPERTIES TIMEOUT 60)
